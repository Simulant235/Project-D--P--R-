# HLE: Systemaufrufe. Implementierungen liegen in <Bibliothek>.cpp, alles noch Fehlende als Log-Stub in stubs/.
file(GLOB DIVA_HLE_STUB_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/stubs/*.cpp")
file(GLOB DIVA_HLE_IMPL_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
list(FILTER DIVA_HLE_IMPL_SOURCES EXCLUDE REGEX "stub_template\.cpp$")
add_library(diva_hle STATIC ${DIVA_HLE_STUB_SOURCES} ${DIVA_HLE_IMPL_SOURCES})
target_compile_features(diva_hle PUBLIC cxx_std_20)
target_link_libraries(diva_hle PUBLIC mfplat mfuuid wmcodecdspuuid ole32 winmm)
target_include_directories(diva_hle PUBLIC "${CMAKE_SOURCE_DIR}/runtime/include")
if(MSVC)
    target_compile_options(diva_hle PRIVATE /W4 /utf-8)
    target_compile_definitions(diva_hle PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
endif()

# ATRAC3plus-Decoder für den Videoton (third_party/at3_standalone, LGPL 2.1, aus PPSSPP/FFmpeg)
file(GLOB AT3_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/third_party/at3_standalone/*.cpp")
add_library(at3_standalone STATIC ${AT3_SOURCES})
target_include_directories(at3_standalone PUBLIC "${CMAKE_SOURCE_DIR}/third_party/at3_standalone" PRIVATE "${CMAKE_SOURCE_DIR}/third_party/at3_standalone/shim")
if(MSVC)
    target_compile_options(at3_standalone PRIVATE /W0)
    target_compile_definitions(at3_standalone PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX _USE_MATH_DEFINES)
endif()
target_link_libraries(diva_hle PUBLIC at3_standalone)
